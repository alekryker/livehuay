name: Update live lottery results

on:
  workflow_dispatch:
  schedule:
    - cron: "*/5 * * * *"

permissions:
  contents: write

concurrency:
  group: live-lottery-update
  cancel-in-progress: true

jobs:
  update:
    runs-on: ubuntu-latest
    timeout-minutes: 12

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: "20"
          cache: npm

      - name: Create package file
        run: |
          if [ ! -f package.json ]; then
            npm init -y
          fi

      - name: Install Playwright
        run: npm install playwright@1.55.0

      - name: Install Chromium
        run: npx playwright install --with-deps chromium

      - name: Fetch live results
        run: node scraper.js

      - name: Commit results.json
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add results.json
          git diff --cached --quiet || git commit -m "Update live results"
          git push
